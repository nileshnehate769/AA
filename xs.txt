  Widget _buildStrategyChip(
    MonitorController controller, {
    required GatewayRoutingMode mode,
    required String label,
    required IconData icon,
  }) {
    final isSelected = controller.gatewayMode == mode;
    return ChoiceChip(
      avatar: Icon(
        icon,
        size: 14,
        color: isSelected ? Colors.black : Colors.grey,
      ),
      label: Text(
        label,
        style: TextStyle(
          fontSize: 11,
          fontWeight: isSelected ? FontWeight.bold : FontWeight.normal,
        ),
      ),
      selected: isSelected,
      selectedColor: controller.accentColor.color,
      onSelected: (_) => controller.setGatewayMode(mode),
    );
  }

  String _getStrategyDescription(GatewayRoutingMode mode, int minutes) {
    return switch (mode) {
      GatewayRoutingMode.fastest => 'Routes all requests through the verified proxy with the lowest response latency.',
      GatewayRoutingMode.roundRobin => 'Changes IP address on every single request. Great for scraping, not for web browsing.',
      GatewayRoutingMode.stickyDomain => 'Locks same-domain requests (e.g. *.github.com) to one IP to keep your login sessions intact.',
      GatewayRoutingMode.stickyTime =>
        'Locks all browsing traffic to the same proxy for $minutes minutes, then rotates to the next.',
    };
  }
