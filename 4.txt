class BrowserPane extends StatefulWidget {
  final String paneId;
  final String appProfile;
  final bool isSplit;
  final VoidCallback onToggleSplit;

  const BrowserPane({
    super.key,
    required this.paneId,
    required this.appProfile,
    required this.isSplit,
    required this.onToggleSplit,
  });

  @override
  State<BrowserPane> createState() => _BrowserPaneState();
}

class _BrowserPaneState extends State<BrowserPane> {
  WebViewEnvironment? _isolatedEnvironment;
  bool _isLoadingEnvironment = true;
  ProxyItem? _activeProxy;

  // --- TAB MANAGEMENT ---
  final List<BrowserTab> _tabs = [];
  int _currentTabIndex = 0;

  // The shared UI controller for the URL bar
  final TextEditingController _urlController = TextEditingController();
  List<BrowserProfile> _profiles = [];

  // Helper to easily get the active tab
  BrowserTab get activeTab => _tabs[_currentTabIndex];

  @override
  void initState() {
    super.initState();
    _addNewTab(); // Create the first tab
    _loadSavedProfiles();
    _initIsolatedEnvironment();
  }

  Future<void> _initIsolatedEnvironment() async {
    // 2. Fetch the proxy once and save it to RAM for instant access!
    _activeProxy = await ProxyManagerService.getActiveProxy(
      widget.paneId,
      widget.appProfile,
    );

    final env = await EnvironmentService.getIsolatedEnvironment(
      widget.paneId,
      widget.appProfile,
    );
    if (mounted) {
      setState(() {
        _isolatedEnvironment = env;
        _isLoadingEnvironment = false;
      });
    }
  }

  Future<void> _loadSavedProfiles() async {
    // Pass widget.appProfile here!
    final profiles = await ProfileService.loadProfiles(widget.appProfile);
    setState(() => _profiles = profiles);
  }

  // --- TAB FUNCTIONS ---
  void _addNewTab([String url = 'https://google.com']) {
    setState(() {
      _tabs.add(
        BrowserTab(
          id: DateTime.now().millisecondsSinceEpoch.toString(),
          url: url,
        ),
      );
      _currentTabIndex = _tabs.length - 1;
      _urlController.text = url;
    });
  }

  void _switchTab(int index) {
    setState(() {
      _currentTabIndex = index;
      _urlController.text = activeTab.url;
    });
  }

  void _closeTab(int index) {
    setState(() {
      if (_tabs.length == 1) {
        // If it's the last tab, reset it to a new blank page
        _tabs[0] = BrowserTab(
          id: DateTime.now().millisecondsSinceEpoch.toString(),
        );
        _urlController.text = _tabs[0].url;
      } else {
        // 1. If we are closing a tab that is BEFORE our active tab,
        // our active tab will shift to the left by 1!
        if (index < _currentTabIndex) {
          _currentTabIndex--;
        }
        // 2. If we are closing the VERY LAST tab while looking at it, shift left
        else if (index == _currentTabIndex && index == _tabs.length - 1) {
          _currentTabIndex--;
        }

        // 3. Now it is safe to remove the tab!
        _tabs.removeAt(index);
        _urlController.text = activeTab.url;
      }
    });
  }

  // --- URL LOADING ---
  void _loadUrl(String input) {
    if (input.isEmpty) return;
    final finalUrl = OmniboxService.parseInput(input);
    activeTab.webViewController?.loadUrl(
      urlRequest: URLRequest(url: WebUri(finalUrl)),
    );
  }

  void _showSaveProfileDialog() {
    final nameController = TextEditingController(text: activeTab.title);
    showDialog(
      context: context,
      builder: (context) {
        return AlertDialog(
          title: const Text('Save Website Profile'),
          content: TextField(
            controller: nameController,
            decoration: const InputDecoration(
              hintText: 'Profile Name (e.g., Work Gmail)',
            ),
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.pop(context),
              child: const Text('Cancel'),
            ),
            TextButton(
              onPressed: () async {
                if (nameController.text.isNotEmpty) {
                  final newProfile = BrowserProfile(
                    id: DateTime.now().millisecondsSinceEpoch.toString(),
                    name: nameController.text,
                    url: _urlController.text,
                    faviconUrl: FaviconService.getFaviconUrl(
                      _urlController.text,
                    ),
                  );
                  await ProfileService.saveProfile(
                    newProfile,
                    widget.appProfile,
                  );
                  await _loadSavedProfiles();
                }
                // ignore: use_build_context_synchronously
                if (mounted) Navigator.pop(context);
              },
              child: const Text('Save'),
            ),
          ],
        );
      },
    );
  }

  // --- APP PROFILE MANAGER LOGIC ---
  void _showProfileManager() async {
    final profiles = await AppProfileService.getProfiles();
    final nameController = TextEditingController();

    if (!mounted) return;
    showDialog(
      context: context,
      builder: (context) {
        return AlertDialog(
          title: Text('Browser Profiles (Current: ${widget.appProfile})'),
          content: SizedBox(
            width: 300,
            child: Column(
              mainAxisSize: MainAxisSize.min,
              children: [
                ...profiles.map(
                  (p) => ListTile(
                    leading: Icon(
                      Icons.account_circle,
                      color: p == widget.appProfile ? Colors.blue : Colors.grey,
                    ),
                    title: Text(p),
                    trailing: p == widget.appProfile
                        ? const Text('Active')
                        : null,
                    onTap: () {
                      Navigator.pop(context);
                      AppProfileService.launchProfileWindow(p);
                    },
                  ),
                ),
                const Divider(),
                TextField(
                  controller: nameController,
                  decoration: const InputDecoration(
                    hintText: 'New Profile Name (e.g. Work)',
                    icon: Icon(Icons.add),
                  ),
                  onSubmitted: (val) async {
                    if (val.isNotEmpty) {
                      await AppProfileService.addProfile(val);
                      // ignore: use_build_context_synchronously
                      Navigator.pop(context);
                      AppProfileService.launchProfileWindow(val);
                    }
                  },
                ),
              ],
            ),
          ),
        );
      },
    );
  }

  // --- JAVASCRIPT INJECTOR UI ---
  void _showScriptEditorDialog() async {
    final currentUrl = activeTab.url;
    final domain = ScriptInjectorService.getDomainFromUrl(currentUrl);

    if (domain.isEmpty) return;

    // Load existing script for this domain if we have one
    final scripts = await ScriptInjectorService.loadScripts(widget.appProfile);
    final existingScript = scripts.where((s) => s.domain == domain).firstOrNull;

    final codeController = TextEditingController(
      text: existingScript?.code ?? '',
    );

    if (!mounted) return;
    showDialog(
      context: context,
      builder: (context) {
        return AlertDialog(
          title: Text('Custom JS for: $domain'),
          content: SizedBox(
            width: 600,
            child: TextField(
              controller: codeController,
              maxLines: 15,
              style: const TextStyle(
                fontFamily: 'Consolas',
                fontSize: 13,
                color: Colors.greenAccent,
              ),
              decoration: const InputDecoration(
                hintText:
                    'Write your JavaScript here...\ne.g. document.body.style.backgroundColor = "black";',
                border: OutlineInputBorder(),
                filled: true,
                fillColor: Colors.black87,
              ),
            ),
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.pop(context),
              child: const Text('Cancel'),
            ),
            TextButton(
              onPressed: () async {
                // FIX: Changed UserScript to CustomScript
                final newScript = CustomScript(
                  domain: domain,
                  code: codeController.text,
                );
                await ScriptInjectorService.saveScript(
                  newScript,
                  widget.appProfile,
                );

                // Immediately run the new script!
                if (activeTab.webViewController != null) {
                  activeTab.webViewController!.evaluateJavascript(
                    source: codeController.text,
                  );
                }

                // FIX: Changed 'mounted' to 'context.mounted' to fix the lint warning
                if (context.mounted) Navigator.pop(context);
              },
              child: const Text(
                'Save & Run',
                style: TextStyle(color: Colors.greenAccent),
              ),
            ),
          ],
        );
      },
    );
  }

                TextButton(
                  onPressed: () => Navigator.pop(context),
                  child: const Text('Cancel'),
                ),
                TextButton(
                  onPressed: () async {
                    setState(() {
                      activeTab.spoofedLat =
                          double.tryParse(latController.text) ??
                          activeTab.spoofedLat;
                      activeTab.spoofedLng =
                          double.tryParse(lngController.text) ??
                          activeTab.spoofedLng;
                    });

                    // Sync the new location and reload the page so the website updates!
                    await LocationSpoofingService.syncLocation(
                      activeTab.webViewController,
                      activeTab.isLocationSpoofed,
                      activeTab.spoofedLat,
                      activeTab.spoofedLng,
                    );
                    activeTab.webViewController?.reload();

                    if (context.mounted) Navigator.pop(context);

  // Automates a seamless 1-second app restart for this specific profile!
  void _restartProfileWindow() {
    // 1. Launch a brand new window for this profile (it will read the new proxy on boot!)
    AppProfileService.launchProfileWindow(widget.appProfile);

    // 2. Kill this current window
    windowManager.close();
  }

  void _showProxyDialog() async {
    final currentProxy = await ProxyManagerService.getActiveProxy(
      widget.paneId,
      widget.appProfile,
    );

    if (!mounted) return;

    // Open our beautiful new UI!
    final result = await showDialog(
      context: context,
      builder: (context) => ProxyManagerDialog(activeProxy: currentProxy),
    );

    // If the user clicked "Turn Off Proxy"
    if (result == "CLEAR") {
      await ProxyManagerService.assignProxy(
        widget.paneId,
        widget.appProfile,
        null,
      );
      _restartProfileWindow();
    }
    // If the user clicked a specific proxy row to assign it
    else if (result is ProxyItem) {
      await ProxyManagerService.assignProxy(
        widget.paneId,
        widget.appProfile,
        result,
      );
      _restartProfileWindow();
    }
  }

  // --- BUILD WEBVIEW FOR A SPECIFIC TAB ---
  Widget _buildWebViewForTab(BrowserTab tab) {
    return InAppWebView(
      key: ValueKey(tab.id),
      webViewEnvironment: _isolatedEnvironment,
      initialUrlRequest: URLRequest(url: WebUri(tab.url)),
      initialSettings: InAppWebViewSettings(
        javaScriptEnabled: true,
        cacheEnabled: true,
        thirdPartyCookiesEnabled: true,
        transparentBackground: true,
        supportMultipleWindows: true,
        useShouldInterceptRequest: true,
        useOnDownloadStart: true,
      ),

      initialUserScripts: UnmodifiableListView<UserScript>([
        MediaSpoofingService.getSpoofingScript(),
        CanvasSpoofingService.getSpoofingScript(),
        ApiRewriterService.getSetupScript(),
        LocationSpoofingService.getSetupScript(),
        AdBlockerService.getCosmeticBlockerScript(),
        WebRTCShieldService.getSetupScript(),
      ]),

      onWebViewCreated: (controller) {
        tab.webViewController = controller;

        // --- LISTEN FOR API LOGS FROM JAVASCRIPT ---
        controller.addJavaScriptHandler(
          handlerName: 'apiSniffer',
          callback: (args) {
            if (args.isNotEmpty) {
              final data = args[0];
              if (mounted) {
                setState(() {
                  tab.apiLogs.add(
                    ApiLog(
                      type: data['type']?.toString() ?? 'UKN',
                      method: data['method']?.toString() ?? 'GET',
                      url: data['url']?.toString() ?? '',
                      status: data['status'] is int ? data['status'] : 0,
                    ),
                  );
                });
              }
            }
          },
        );

        // --- LISTEN FOR THE ZAPPER FINISHING ---
        controller.addJavaScriptHandler(
          handlerName: 'zapperFinished',
          callback: (args) async {
            if (args.isNotEmpty) {
              final data = args[0];
              final domain = ElementZapperService.getDomainFromUrl(tab.url);

              final newZap = ZappedElement(
                id: DateTime.now().millisecondsSinceEpoch.toString(),
                domain: domain,
                selector: data['selector'],
                name: data['name'],
              );

              // Save it to hard drive
              await ElementZapperService.saveZap(newZap, widget.appProfile);

              if (mounted) {
                setState(() {
                  tab.isZappingCurrently = false; // Turn off crosshair
                  tab.activeZaps.add(newZap); // sidebar
                });
                // Instantly hide it permanently!
                ElementZapperService.applySavedZaps(
                  tab.webViewController,
                  tab.activeZaps, 
                );
              }
            }
          },
        );
      },

      // --- POPUP & NEW WINDOW HANDLER ---
      onCreateWindow: (controller, createWindowAction) async {
        final urlToOpen = createWindowAction.request.url;

        if (urlToOpen != null && urlToOpen.toString().isNotEmpty) {
          // 1. Grab the hidden URL the website was trying to open
          final newUrl = urlToOpen.toString();

          // 2. Spawn a brand new tab inside our app with that URL!
          if (mounted) {
            _addNewTab(newUrl);
          }
        }

        // 3. Return TRUE to tell the C++ engine "Don't crash, we handled the popup natively!"
        return true;
      },
      // ----------------------------------

      // --- PROXY & HTTP AUTHENTICATION INTERCEPTOR ---
      // --- PROXY & HTTP AUTHENTICATION INTERCEPTOR ---
      // --- PROXY & HTTP AUTHENTICATION INTERCEPTOR ---
      onReceivedHttpAuthRequest: (controller, challenge) async {
        // 3. Use the instantly available RAM variable instead of waiting for the hard drive!
        if (_activeProxy != null &&
            _activeProxy!.username.isNotEmpty &&
            _activeProxy!.password.isNotEmpty) {
          debugPrint("Injecting Proxy Auth for ${_activeProxy!.hostPort}...");
          return HttpAuthResponse(
            username: _activeProxy!.username.trim(),
            password: _activeProxy!.password.trim(),
            action: HttpAuthResponseAction.PROCEED,
          );
        }

        return HttpAuthResponse(action: HttpAuthResponseAction.CANCEL);
      },
      // -----------------------------------------------
      // -----------------------------------------------

      // 1. Silently grant general permissions (Camera, Microphone, etc.)
      onPermissionRequest: (controller, request) async {
        return PermissionResponse(
          resources: request.resources,
          action: PermissionResponseAction.GRANT,
        );
      },

      // 2. Silently grant Geolocation requests so Windows doesn't panic
      onGeolocationPermissionsShowPrompt: (controller, origin) async {
        return GeolocationPermissionShowPromptResponse(
          origin: origin,
          allow: true,
          retain: true,
        );
      },

      onTitleChanged: (controller, title) {
        if (title != null) setState(() => tab.title = title);
      },

      shouldInterceptRequest: (controller, request) async {
        // We removed the '!= null' check because it is guaranteed to be safe!
        final urlString = request.url.toString();

        // 1. ADVANCED AD BLOCKER
        if (AdBlockerService.isAdUrl(urlString)) {
          return WebResourceResponse(
            contentType: 'text/plain',
            data: Uint8List.fromList([]),
            statusCode: 403,
            reasonPhrase: 'Blocked by AdBlocker',
          );
        }

        // 2. RAW MEDIA SNIFFER!
        final lowerUrl = urlString.toLowerCase();
        if (lowerUrl.contains('.mp4') ||
            lowerUrl.contains('.webm') ||
            lowerUrl.contains('.m3u8')) {
          if (!tab.detectedMediaUrls.contains(urlString)) {
            // Safely add it to the UI state without breaking the background thread
            Future.microtask(() {
              if (mounted) setState(() => tab.detectedMediaUrls.add(urlString));
            });
          }
        }

        return null;
      },

      // ---------------------------------------
      onFindResultReceived:
          (controller, activeMatchOrdinal, numberOfMatches, isDoneCounting) {
            if (mounted) {
              setState(() {
                tab.currentMatch = activeMatchOrdinal;
                tab.totalMatches = numberOfMatches;
              });
            }
          },
      onLoadStart: (controller, url) async {
        setState(() {
          tab.detectedMediaUrls.clear();
        });

        controller.evaluateJavascript(
          source: ApiSnifferService.injectionScript,
        );

        setState(() => tab.isZapperActive = false);

        // --- SYNC THE REWRITE RULES INTO THE PAGE ---
        await ApiRewriterService.syncRulesToBrowser(
          controller,
          widget.appProfile,
        );

        // --- SYNC GPS LOCATION TO BROWSER ---
        await LocationSpoofingService.syncLocation(
          controller,
          tab.isLocationSpoofed,
          tab.spoofedLat,
          tab.spoofedLng,
        );

        await WebRTCShieldService.syncShield(
          controller,
          tab.isWebRTCShieldEnabled,
        );

        if (url != null) {
          setState(() {
            tab.url = url.toString();
            if (_tabs[_currentTabIndex].id == tab.id) {
              _urlController.text = tab.url;
            }
          });
        }
      },

      onLoadStop: (controller, url) async {
        if (url != null) {
          setState(() {
            tab.url = url.toString();
            if (_tabs[_currentTabIndex].id == tab.id) {
              _urlController.text = tab.url;
            }
          });
        }
        await ZoomService.applyZoom(controller, tab.zoom);

        await ScriptInjectorService.injectScripts(
          controller,
          tab.url,
          widget.appProfile,
        );

        await controller.evaluateJavascript(
          source: ElementZapperService.setupScript,
        );
        final domain = ElementZapperService.getDomainFromUrl(tab.url);
        final allZaps = await ElementZapperService.loadZaps(widget.appProfile);

        if (mounted) {
          setState(() {
            // Filter the database to only show Zaps for the current website
            tab.activeZaps = allZaps.where((z) => z.domain == domain).toList();
          });
          // Inject the CSS to hide them immediately!
          await ElementZapperService.applySavedZaps(controller, tab.activeZaps);
        }
      },
      onProgressChanged: (controller, progress) {
        setState(() => tab.progress = progress / 100);
      },

      onDownloadStartRequest: (controller, downloadRequest) async {
        await DownloadManagerService.handleDownload(downloadRequest);
      },
    );
  }

  @override
  Widget build(BuildContext context) {
    // 1. Wrap the entire pane in a Focus widget so it listens to the keyboard
    return Focus(
      autofocus: true,
      // 2. Map your keyboard shortcuts to your functions!
      child: CallbackShortcuts(

                          size: 30,
                        ),

                        const Expanded(
                          child: Text(
                            "YouTube Video Detected!",
                            style: TextStyle(fontWeight: FontWeight.bold),

                        ),

                    "No raw video files detected in network traffic.",

                    shrinkWrap: true,
