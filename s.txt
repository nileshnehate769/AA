  /// Adds files to the share hub via Windows File Picker
  Future<void> pickAndAddSharedFiles() async {
    try {
      final result = await FilePicker.platform.pickFiles(
        allowMultiple: true,
        dialogTitle: 'Select Files to Share over Wi-Fi',
      );

      if (result != null && result.files.isNotEmpty) {
        for (var file in result.files) {
          if (file.path != null) {
            final f = File(file.path!);
            if (f.existsSync()) {
              final ext = file.extension?.toLowerCase() ?? '';
              final isMedia = [
                'mp4',
                'mkv',
                'webm',
                'mp3',
                'wav',
                'ogg',
                'avi',
              ].contains(ext);

              _wifiService.sharedFiles.add(
                SharedFileItem(
                  id: DateTime.now().microsecondsSinceEpoch.toString(),
                  name: file.name,
                  path: file.path!,
                  sizeBytes: file.size,
                  extension: ext,
                  isMedia: isMedia,
                ),
              );
            }
          }
        }
        addLog(
          'Shared ${_wifiService.sharedFiles.length} file(s) on Wi-Fi hub.',
        );
        notifyListeners();
      }
    } catch (e) {
      addLog('Error picking files: $e');
    }
  }

  /// 1-Click: Warms up the top fastest active nodes to eliminate cold-start delays
  Future<void> warmUpActivePool({int topCount = 10}) async {
    if (isWarmingUp) return;

    final activeNodes = _dbService.getActiveProxiesForGateway(
      currentProfileName,
    );
    if (activeNodes.isEmpty) {
      addLog(
        'Cannot warm up pool: No active proxies found in current workspace.',
      );
      return;
    }

    isWarmingUp = true;
    notifyListeners();

    final targetNodes = activeNodes.take(topCount).toList();
    addLog(
      'Warming up top ${targetNodes.length} fastest active nodes (TCP Pre-Handshake)...',
    );

    final result = await PoolWarmupService.warmUpNodes(
      targetNodes,
      timeoutSeconds: 3,
    );

    // Persist freshly primed latencies into SQLite
    for (final node in targetNodes) {
      _dbService.updateProxyRecord(currentProfileName, node);
    }
    refreshStats();

    isWarmingUp = false;
    addLog(
      '[Pool Warm-Up] Primed ${result.primedCount}/${targetNodes.length} nodes in ${result.elapsedTotalMs}ms. '
      'Fastest: ${result.fastestLatencyMs ?? 0}ms. Sockets ready for instant browsing!',
    );

    notifyListeners();
  }

  List<GatewayTrafficEntry> get filteredTraffic {

    return liveTraffic

  /// Circuit Breaker: Called when an active proxy successfully handles a live request
  void handleGatewayNodeSuccess(ProxyRecord proxy, int liveDurationMs) {
    // 1. Reset consecutive failures
    _consecutiveFailures[proxy.address] = 0;

    // 2. Severe Latency Check (> 8000ms triggers circuit breaker)
    if (liveDurationMs >= 8000) {
      _tripCircuitBreaker(
        proxy,
        reason:
            'Severe latency spike (${(liveDurationMs / 1000).toStringAsFixed(1)}s > 8s limit)',
      );
      return;
    }

    // 3. Smooth rolling latency update (60% historical + 40% live)
    if (proxy.latencyMs != null && liveDurationMs > 0) {
      proxy.latencyMs = ((proxy.latencyMs! * 0.6) + (liveDurationMs * 0.4))
          .round();
      _dbService.updateProxyRecord(currentProfileName, proxy);
    }
  }

  /// Circuit Breaker: Called when a live request times out or is rejected
  void handleGatewayNodeDropped(ProxyRecord proxy, int liveDurationMs) {
    final fails = (_consecutiveFailures[proxy.address] ?? 0) + 1;
    _consecutiveFailures[proxy.address] = fails;

    // RULE: 2 consecutive failures OR request took > 8000ms before failing
    if (fails >= 2 || liveDurationMs >= 8000) {
      _tripCircuitBreaker(
        proxy,
        reason: fails >= 2
            ? '2 consecutive live request failures'
            : 'Timed out after 8s',
      );
    } else {
      // First temporary hiccup: add penalty latency so faster nodes are prioritized first
      proxy.latencyMs = (proxy.latencyMs ?? 1000) + 2000;
      _dbService.updateProxyRecord(currentProfileName, proxy);
      refreshStats();
      notifyListeners();
    }
  }

  /// Demotes node in SQLite, removes it from gateway rotation, and logs alert
  void _tripCircuitBreaker(ProxyRecord proxy, {required String reason}) {
    totalCircuitBreakerTrips++;
    proxy.status = 'Failed (Circuit Breaker)';
    _consecutiveFailures.remove(proxy.address);

    // Update SQLite immediately so getActiveProxiesForGateway() stops returning this node
    _dbService.updateProxyRecord(currentProfileName, proxy);
    refreshStats();

    addLog(
      '[CIRCUIT BREAKER] Tripped on ${proxy.address}: $reason. Node auto-demoted to Failed.',
    );
    notifyListeners();
    SystemTrayService().updateTrayMenu();
  }

  // Deduplication & Import Engine
  Future<void> importRawLines(
    List<String> lines, {
    String sourceName = 'Source',
  }) async {
    if (lines.isEmpty) return;

    addLog('Parsing ${lines.length} lines in background isolate...');

    final List<ParsedProxyDto> parsed = await Isolate.run(() {
      final List<ParsedProxyDto> results = [];
      final seen = <String>{};

      for (var rawLine in lines) {
        var line = rawLine.trim();
        if (line.isEmpty) continue;

        line = line.replaceAll(
          RegExp(r'^(http|https|socks4|socks5)://', caseSensitive: false),
          '',
        );
        if (line.endsWith('/')) line = line.substring(0, line.length - 1);

        String? user;
        String? pass;
        String address = '';

        if (line.contains('@')) {
          final parts = line.split('@');
          final creds = parts[0].split(':');
          if (creds.length == 2) {
            user = creds[0];
            pass = creds[1];
          }
          address = parts[1].trim();
        } else {
          final parts = line.split(':');
          if (parts.length == 4) {
            address = '${parts[0].trim()}:${parts[1].trim()}';
            user = parts[2].trim();
            pass = parts[3].trim();
          } else if (parts.length == 2) {
            address = line.trim();
          }
        }

        if (address.isNotEmpty && address.contains(':')) {
          final key = '${address.toLowerCase()}:${user ?? ""}';
          if (!seen.contains(key)) {
            seen.add(key);
            results.add(
              ParsedProxyDto(address: address, username: user, password: pass),
            );
          }
        }
      }
      return results;
    });

    final newRecords = parsed
        .map(
          (p) => ProxyRecord(
            address: p.address,
            username: p.username,
            password: p.password,
          ),
        )
        .toList();

    // High-speed SQLite transaction insert
    final inserted = _dbService.batchInsertProxies(
      currentProfileName,
      newRecords,
    );
    refreshStats();
    addLog('Stored $inserted new unique nodes into SQLite from $sourceName.');
    currentPage = 1;
    notifyListeners();
  }

  /// Toggles the local HTTP/HTTPS gateway on 127.0.0.1:[gatewayPort]
  /// Toggles the Dual Gateway (HTTP on 8888 & SOCKS5 on 1080)
  Future<void> toggleGateway() async {
    if (isGatewayRunning || isSocks5Running) {
      if (isSystemProxyActive) await toggleSystemProxy();
      await _gatewayService.stopServer();
      await _socks5Service.stopServer(); // Stop SOCKS5
      addLog('Local gateway & SOCKS5 bridge stopped.');
    } else {
      final activeNodes = _dbService.getActiveProxiesForGateway(
        currentProfileName,
      );
      if (activeNodes.isEmpty) {
        addLog(
          'Cannot start gateway: 0 active proxies found in SQLite database.',
        );
        return;
      }

      addLog(
        'Gateway loaded ${activeNodes.length} active proxies from SQLite.',
      );

      // 1. Start HTTP / HTTPS Gateway on 8888
      // 1. Start HTTP Gateway with Geo-Fence active
      await _gatewayService.startServer(
        port: gatewayPort,
        getActiveProxies: () => _dbService.getActiveProxiesForGateway(
          currentProfileName,
          excludedCountries: excludedCountries.toList(),
          excludeSpam: excludeSpamFromGateway,
          excludeTampered: excludeTamperedFromGateway,
        ),

        mode: gatewayMode,
        stickyMinutes: stickyMinutes,
        isMultiHop: isMultiHopEnabled,
        exitCountry: multiHopExitCountry,
        killSwitch: isKillSwitchEnabled,
        httpsOnly: isHttpsOnlyEnabled,
        requireAuth: isGatewayAuthEnabled,
        username: gatewayUsername,
        password: gatewayPassword,
        enableShield: isDomainShieldEnabled,
        enableHttpsInspection: isHttpsInspectionEnabled,
        onLog: addLog,
        onNodeDropped: (droppedNode, duration) {
          droppedNode.latencyMs = (droppedNode.latencyMs ?? 1000) + 2500;
          _dbService.updateProxyRecord(currentProfileName, droppedNode);
          refreshStats();
          notifyListeners();
        },
        onNodeSuccess: handleGatewayNodeSuccess,
        onTraffic: recordTrafficEntry,
      );

      // 2. Start SOCKS5 Gateway with Geo-Fence active
      await _socks5Service.startServer(
        port: socks5Port,
        getActiveProxies: () => _dbService.getActiveProxiesForGateway(
          currentProfileName,
          excludedCountries: excludedCountries.toList(),
        ),
        onLog: addLog,
        onNodeDropped: (droppedNode) {
          droppedNode.latencyMs = (droppedNode.latencyMs ?? 1000) + 2500;
          _dbService.updateProxyRecord(currentProfileName, droppedNode);
          refreshStats();
          notifyListeners();
        },
        onTraffic: recordTrafficEntry,
        killSwitch: isKillSwitchEnabled,
        enableShield: isDomainShieldEnabled,
      );
    }
    notifyListeners();
    SystemTrayService().updateTrayMenu();
  }

      if (usePacMode) {
        // Smart PAC Mode
        final pacUrl = 'http://127.0.0.1:$gatewayPort/proxy.pac';

          );
        }
      } else {
        // Global Tunnel Mode
        ok = await _systemProxyService.enableSystemProxy(
          '127.0.0.1:$gatewayPort',
        );
        if (ok) {
          isSystemProxyActive = true;
          addLog(
            'Windows Proxy active via Global Tunnel (127.0.0.1:$gatewayPort).',
          );
        }
      }
    }
    notifyListeners();
    SystemTrayService().updateTrayMenu();
  }

  /// Re-test a single endpoint immediately
  Future<void> retestSingleProxy(ProxyRecord record) async {
    record.status = 'Testing...';
    notifyListeners();

    final result = await _networkService.runDiagnosticInIsolate(
      address: record.address,
      username: record.username,
      password: record.password,
      realClientIp: localPublicIp,
      baselineDnsIp: baselineDnsIp,
      timeoutSeconds: timeoutSeconds,
    );

    record.status = result.status;
    record.latencyMs = result.latencyMs;
    record.tcpPingMs = result.tcpPingMs;
    record.protocol = result.protocol;
    record.anonymity = result.anonymity;
    record.sslSupported = result.sslSupported;
    record.isDnsLeaked = result.isDnsLeaked;
    record.dnsServerIp = result.dnsServerIp;

    if (record.status == 'Active') {
      // 1. GeoIP Lookup (Borrow active proxy from SQLite if rate-limited)
      final host = record.address.split(':')[0];
      final geo = await _geoIpService.lookupCountry(
        host,
        getActiveProxies: () =>
            _dbService.getActiveProxiesForGateway(currentProfileName),
        onLog: addLog,
      );
      if (geo != null) {
        record.country = geo['country'];
        record.countryCode = geo['countryCode'];
        record.isp = geo['isp'];
      }

      final abuseResult = await AbuseIpdbService().checkIpAbuse(
        host,
        onLog: addLog,
        onKeyUpdated: reloadAbuseKeys,
      );
      if (abuseResult != null) {
        record.abuseScore = abuseResult.abuseScore;
        record.totalAbuseReports = abuseResult.totalReports;
        record.ipUsageType = abuseResult.usageType;
      }

      // 2. Automated SSL/MitM & Certificate Transparency Audit
      if (autoAuditSsl) {
        final sslResult = await SslIntegrityService.checkCertificateIntegrity(
          record,
          timeoutSeconds: 3,
        );
        record.isMitmDetected = sslResult.isMitmDetected;
        record.sslCertificateIssuer = sslResult.issuer;
        record.isCtCompliant = sslResult.isCtCompliant;
        record.sctCount = sslResult.sctCount;
      }

      // 3. Automated Platform Reachability (Google, Cloudflare, Wikipedia, GitHub)
      if (autoProbePlatforms) {
        final reachability = await PlatformProbeService.probePlatforms(
          record,
          timeoutSeconds: 3,
        );
        record.platformReachability = reachability;
      }

      // 4. Automated ECH / SNI Privacy Audit
      if (autoAuditEch) {
        final echResult = await EchInspectorService.checkEchReadiness(
          record,
          timeoutSeconds: 3,
        );
        record.isEchReady = echResult.isEchReady;
        record.echDetails = echResult.details;
      }
      if (autoProbeSla && record.status == 'Active') {
        final slaResult = await SlaProbeService.auditRegionalSla(
          record,
          timeoutSeconds: 3,
        );
        record.jitterMs = slaResult.jitterMs;
        record.slaScore = slaResult.slaScore;
        record.regionalLatencies = slaResult.regionalLatencies;
      }

      if (autoSpeedTest && record.status == 'Active') {
        final speed = await SpeedTestService.measureDownloadSpeed(
          record,
          timeoutSeconds: 5,
        );
        record.downloadSpeedBps = speed;
      }
      if (autoAuditTampering) {
        final tamperResult =
            await ContentIntegrityService.auditContentIntegrity(
              record,
              timeoutSeconds: 3,
            );
        record.isContentTampered = tamperResult.isTampered;
        record.tamperDetails = tamperResult.details;
      }
    }

    // =========================================================================
    // CRITICAL: Save to SQLite database and refresh dashboard stats!
    // =========================================================================
    _dbService.updateProxyRecord(currentProfileName, record);
    refreshStats();

    addLog(
      'Retested ${record.address}: ${record.status} (${record.latencyMs ?? 0}ms)',
    );

    notifyListeners();
  }

  /// Creates a new workspace (clean or cloned)
  Future<void> createProfile(
    String newName, {
    bool cloneCurrent = false,
  }) async {
    final trimmed = newName.trim();
    if (trimmed.isEmpty || availableProfiles.contains(trimmed)) return;

    // Save current workspace first
    await saveCurrentProfile();

    final newProfile = ProfileData(
      name: trimmed,
      targetUrl: cloneCurrent ? targetUrl : 'https://www.example.com',
      timeoutSeconds: cloneCurrent ? timeoutSeconds : 5,
      concurrency: cloneCurrent ? concurrency : 4,
      gatewayPort: cloneCurrent ? gatewayPort : 8888,
      gatewayMode: cloneCurrent ? gatewayMode : GatewayRoutingMode.fastest,
      proxies: cloneCurrent ? List<ProxyRecord>.from(proxies) : [],
    );

    await _storageService.saveProfile(newProfile);
    availableProfiles = await _storageService.getProfileNames();
    await switchProfile(trimmed);
  }

    final oldName = currentProfileName;
    await _storageService.renameProfile(oldName, trimmed);
    currentProfileName = trimmed;
    availableProfiles = await _storageService.getProfileNames();
    addLog('Renamed workspace "$oldName" to "$trimmed".');
    notifyListeners();
  }

    if (availableProfiles.length <= 1) {

    final toDelete = currentProfileName;
    availableProfiles.remove(toDelete);
    await _storageService.deleteProfile(toDelete);

    // Switch to first remaining profile
    final nextProfile = availableProfiles.first;
    await switchProfile(nextProfile);
    addLog('Deleted workspace "$toDelete". Switched to "$nextProfile".');
  }

  Future<void> _runLoop() async {
    while (isLooping) {
      if (!isRunning && totalCount > 0) {

  Future<void> runDiagnostics({int concurrency = 4}) async {
    if (isRunning) return;

    // Fetch candidate proxies from SQLite (skipping Failed)
    final queue = _dbService.getProxiesToTest(currentProfileName);

    if (queue.isEmpty) {
      addLog(
        'All nodes are marked as Failed ($failedCount skipped). Right-click a node to retest.',
      );
      return;
    }

    final int targetCount = queue.length;
    final int skippedCount = totalCount - targetCount;

    isRunning = true;
    progress = 0.0;
    addLog(
      'Starting test on $targetCount nodes ($skippedCount failed skipped)...',
    );
    notifyListeners();

    if (localPublicIp == null) {
      addLog('Analyzing local network baseline...');
      localPublicIp = await _networkService.getPublicIp();
      baselineDnsIp = await _networkService.getBaselineDnsIp();
    }

    int completed = 0;
    int queueIndex = 0;
    DateTime lastUiUpdate = DateTime.now();

    Future<void> worker() async {
      while (queueIndex < queue.length && isRunning) {
        final record = queue[queueIndex++];

        try {
          final result = await _networkService.runDiagnosticInIsolate(
            address: record.address,
            username: record.username,
            password: record.password,
            realClientIp: localPublicIp,
            baselineDnsIp: baselineDnsIp,
            timeoutSeconds: timeoutSeconds,
          );

          record.status = result.status;
          record.latencyMs = result.latencyMs;
          record.tcpPingMs = result.tcpPingMs;
          record.protocol = result.protocol;
          record.anonymity = result.anonymity;
          record.sslSupported = result.sslSupported;
          record.isDnsLeaked = result.isDnsLeaked;
          record.dnsServerIp = result.dnsServerIp;

          record.isMitmDetected = result.isMitmDetected;
          record.sslCertificateIssuer = result.sslCertificateIssuer;

          if (record.isMitmDetected) {
            addLog(
              '[THREAT ALERT] Proxy ${record.address} is intercepting SSL traffic (${record.sslCertificateIssuer}). Node blocked.',
            );
          }

          if (record.status == 'Active') {
            final host = record.address.split(':')[0];
            final geo = await _geoIpService.lookupCountry(
              host,
              getActiveProxies: () =>
                  _dbService.getActiveProxiesForGateway(currentProfileName),
              onLog: addLog,
            );
            if (geo != null) {
              record.country = geo['country'];
              record.countryCode = geo['countryCode'];
              record.isp = geo['isp'];
            }

            if (record.status == 'Active') {
              final host = record.address.split(':')[0];

              // 1. GeoIP Lookup
              final geo = await _geoIpService.lookupCountry(
                host,
                getActiveProxies: () =>
                    _dbService.getActiveProxiesForGateway(currentProfileName),
                onLog: addLog,
              );
              if (geo != null) {
                record.country = geo['country'];
                record.countryCode = geo['countryCode'];
                record.isp = geo['isp'];
              }

              // =================================================================
              // AUTOMATED ABUSEIPDB FRAUD & SPAM SCORE LOOKUP (Auto-detect)
              // =================================================================
              final abuseResult = await AbuseIpdbService().checkIpAbuse(
                host,
                onLog: addLog,
                onKeyUpdated: reloadAbuseKeys,
              );
              if (abuseResult != null) {
                record.abuseScore = abuseResult.abuseScore;
                record.totalAbuseReports = abuseResult.totalReports;
                record.ipUsageType = abuseResult.usageType;

                if (record.isSpam) {
                  addLog(
                    '[AbuseIPDB Flag] ${record.address} has ${record.abuseScore}% Abuse Confidence Score. Flagged as SPAM.',
                  );
                }
              }

              // =================================================================
              // AUTOMATED SSL/MitM & CERTIFICATE TRANSPARENCY AUDIT
              // =================================================================
              if (autoAuditSsl) {
                final sslResult =
                    await SslIntegrityService.checkCertificateIntegrity(
                      record,
                      timeoutSeconds: 3,
                    );
                record.isMitmDetected = sslResult.isMitmDetected;
                record.sslCertificateIssuer = sslResult.issuer;
                record.isCtCompliant = sslResult.isCtCompliant;
                record.sctCount = sslResult.sctCount;
              }

              // Platform reachability audit (if enabled)
              if (autoProbePlatforms && record.status == 'Active') {
                final reachability = await PlatformProbeService.probePlatforms(
                  record,
                  timeoutSeconds: 3,
                );
                record.platformReachability = reachability;
              }

              // AUTOMATED ECH & SNI PRIVACY AUDIT (NEW!)
              if (autoAuditEch && record.status == 'Active') {
                final echResult = await EchInspectorService.checkEchReadiness(
                  record,
                  timeoutSeconds: 3,
                );
                record.isEchReady = echResult.isEchReady;
                record.echDetails = echResult.details;
              }

              // Automated Multi-Region SLA & Jitter Audit (if enabled)
              if (autoProbeSla && record.status == 'Active') {
                final slaResult = await SlaProbeService.auditRegionalSla(
                  record,
                  timeoutSeconds: 3,
                );
                record.jitterMs = slaResult.jitterMs;
                record.slaScore = slaResult.slaScore;
                record.regionalLatencies = slaResult.regionalLatencies;
              }

              // =================================================================
              // AUTOMATED HTML CONTENT INTEGRITY & AD-INJECTION AUDIT
              // =================================================================
              if (autoAuditTampering && record.status == 'Active') {
                final tamperResult =
                    await ContentIntegrityService.auditContentIntegrity(
                      record,
                      timeoutSeconds: 3,
                    );
                record.isContentTampered = tamperResult.isTampered;
                record.tamperDetails = tamperResult.details;

                if (record.isContentTampered) {
                  addLog(
                    '[TAMPERING DETECTED] ${record.address} injected scripts/ads into HTTP traffic! (${tamperResult.details})',
                  );
                }
              }

              // =================================================================
              // AUTOMATED DOWNLOAD SPEED BENCHMARK (Only if toggle is enabled!)
              // =================================================================
              if (autoSpeedTest && record.status == 'Active') {
                final speedBps = await SpeedTestService.measureDownloadSpeed(
                  record,
                  timeoutSeconds: 5,
                );
                record.downloadSpeedBps = speedBps;
                if (speedBps != null) {
                  addLog(
                    '[Speed Benchmark] ${record.address} download throughput: ${record.formattedSpeed}',
                  );
                }
              }
            }
          }
        } catch (_) {
          record.status = 'Failed';
        }

        // Save verification update to SQLite immediately
        _dbService.updateProxyRecord(currentProfileName, record);

        completed++;
        progress = targetCount > 0
            ? (completed / targetCount).clamp(0.0, 1.0)
            : 0.0;

        final now = DateTime.now();
        if (now.difference(lastUiUpdate).inMilliseconds > 60 ||
            completed == targetCount) {
          lastUiUpdate = now;
          refreshStats();
          notifyListeners();
        }
      }
    }

    final workers = List.generate(
      concurrency > queue.length ? queue.length : concurrency,
      (_) => worker(),
    );

    await Future.wait(workers);

    isRunning = false;
    refreshStats();
    addLog('Diagnostics complete. SQLite updated.');
    notifyListeners();
  }

  Future<void> startPortScan(String host, {List<int>? ports}) async {
    if (isPortScanning) return;

    final cleanHost = host
        .trim()
        .replaceAll(RegExp(r'^(https?:\/\/|socks5:\/\/)'), '')
        .split(':')
        .first;
    if (cleanHost.isEmpty) return;

    isPortScanning = true;
    portScanTargetHost = cleanHost;
    portScanResults.clear();
    addLog('[Port Scanner] Starting scan on $cleanHost...');
    notifyListeners();

    try {
      await _portScanner
          .scanHost(
            host: cleanHost,
            targetPorts: ports,
            onPortScanned: (result) {
              // Updates UI live as each port finishes probing
              notifyListeners();
            },
          )
          .then((list) {
            portScanResults = list;
            final openCount = list.where((r) => r.isOpen).length;
            addLog(
              '[Port Scanner] Finished scan on $cleanHost: Found $openCount open ports.',
            );
          });
    } catch (e) {
      addLog('[Port Scanner] Error scanning $cleanHost: $e');
    } finally {
      isPortScanning = false;
      notifyListeners();
    }
  }

  void clearPortScanResults() {
    portScanResults.clear();
    portScanTargetHost = '';
    notifyListeners();
  }

  /// Exports recorded Live Traffic entries into a W3C HAR 1.2 file
  Future<bool> exportTrafficHar() async {
    if (liveTraffic.isEmpty) {
      addLog('No traffic recorded to export.');
      return false;
    }

    addLog('Exporting ${liveTraffic.length} traffic entries to HAR archive...');
    final success = await HarExportService.exportToHar(liveTraffic);
    if (success) {
      addLog('HAR file saved successfully.');
    }
    return success;
  }

  /// Generates and prompts to save the Support & Diagnostic Bundle
  Future<bool> generateSupportBundle() async {
    addLog('Generating system diagnostic and support bundle...');
    return await SupportBundleService.generateAndExportReport(this);
  }

  /// 1-Click: Clears flapping penalty for a specific node
  void resetFlapHistory(ProxyRecord record) {
    record.flapCount = 0;
    record.lastStatusChange = null;
    _dbService.resetProxyFlapCount(currentProfileName, record);
    refreshStats();
    addLog(
      'Reset flap penalty for ${record.address}. Node restored to 100% stability.',
    );
    notifyListeners();
  }

  void stop() async {
    isRunning = false;
    addLog('Execution stopped by user. Saving current state...');
    await saveCurrentProfile();
    notifyListeners();
  }

  @override
  void dispose() {
    NetworkHandoverService().stopMonitoring();
    if (isSystemProxyActive) {
      _systemProxyService.disableSystemProxy();
    }
    _gatewayService.stopServer();
    _socks5Service.stopServer();
    _wifiService.stopServer();
    super.dispose();
  }
}
